function trimWhite(cv){
 try{
  const ctx=cv.getContext('2d'),w=cv.width,h=cv.height;
  const d=ctx.getImageData(0,0,w,h).data;
  const bg=i=>d[i]>238&&d[i+1]>238&&d[i+2]>238;
  const rowBg=y=>{for(let x=0;x<w;x+=2){if(!bg((y*w+x)*4))return false}return true};
  const colBg=x=>{for(let y=0;y<h;y+=2){if(!bg((y*w+x)*4))return false}return true};
  let top=0,bot=h-1,left=0,right=w-1;
  while(top<bot&&rowBg(top))top++;
  while(bot>top&&rowBg(bot))bot--;
  while(left<right&&colBg(left))left++;
  while(right>left&&colBg(right))right--;
  const pad=Math.round(Math.max(w,h)*0.015);
  left=Math.max(0,left-pad);top=Math.max(0,top-pad);
  right=Math.min(w-1,right+pad);bot=Math.min(h-1,bot+pad);
  const nw=right-left+1,nh=bot-top+1;
  if((nw<w*0.98||nh<h*0.98)&&nw>40&&nh>40){
   const c2=document.createElement('canvas');c2.width=nw;c2.height=nh;
   c2.getContext('2d').drawImage(cv,left,top,nw,nh,0,0,nw,nh);
   return c2;
  }
 }catch(e){}
 return cv;
}
/* otomatik iyilestirme: kontrast esitleme + hafif canlilik + netlestirme */
function enhanceCanvas(cv){
 try{
  const ctx=cv.getContext('2d');
  const im=ctx.getImageData(0,0,cv.width,cv.height);
  const d=im.data,total=d.length/4;
  const hist=new Array(256).fill(0);
  for(let i=0;i<d.length;i+=4)hist[(d[i]*299+d[i+1]*587+d[i+2]*114)/1000|0]++;
  let lo=0,hi=255,acc=0;
  for(let i=0;i<256;i++){acc+=hist[i];if(acc>total*0.01){lo=i;break}}
  acc=0;for(let i=255;i>=0;i--){acc+=hist[i];if(acc>total*0.01){hi=i;break}}
  if(hi>lo+12){
   const k=255/(hi-lo);
   for(let i=0;i<d.length;i+=4){
    for(let c=0;c<3;c++){const v=(d[i+c]-lo)*k;d[i+c]=v<0?0:v>255?255:v}
    const r=d[i],g=d[i+1],b=d[i+2],m=(r+g+b)/3;
    d[i]=Math.min(255,Math.max(0,m+(r-m)*1.1));
    d[i+1]=Math.min(255,Math.max(0,m+(g-m)*1.1));
    d[i+2]=Math.min(255,Math.max(0,m+(b-m)*1.1));
   }
  }
  /* hafif keskinlestirme (unsharp) */
  const w=cv.width,h=cv.height,src=new Uint8ClampedArray(d),a=0.28;
  for(let y=1;y<h-1;y++){
   for(let x=1;x<w-1;x++){
    const i=(y*w+x)*4;
    for(let c=0;c<3;c++){
     const v=src[i+c]*(1+4*a)-a*(src[i+c-4]+src[i+c+4]+src[i+c-w*4]+src[i+c+w*4]);
     d[i+c]=v<0?0:v>255?255:v;
    }
   }
  }
  ctx.putImageData(im,0,0);
  /* koseye silinemez ÜÇEL filigrani */
  const TXT='ÜÇEL STEEL CONSTRUCTION';
  let fs=Math.max(10,cv.width/42);
  ctx.font='700 '+fs+'px Arial';
  const maxw=cv.width*0.45;
  let tw=ctx.measureText(TXT).width;
  if(tw>maxw){fs=Math.max(8,fs*maxw/tw);ctx.font='700 '+fs+'px Arial'}
  ctx.textAlign='right';ctx.textBaseline='bottom';
  ctx.lineWidth=Math.max(1,fs/10);
  ctx.strokeStyle='rgba(0,0,0,.30)';ctx.fillStyle='rgba(255,255,255,.65)';
  ctx.strokeText(TXT,cv.width-fs*0.8,cv.height-fs*0.6);
  ctx.fillText(TXT,cv.width-fs*0.8,cv.height-fs*0.6);
 }catch(e){}
}
function shrink(file,cb){

 r.onload=e=>{
  const im=new Image();
  im.onload=()=>{
   const k=Math.min(1,2400/Math.max(im.width,im.height));
   const cv=document.createElement('canvas');
   cv.width=Math.round(im.width*k);cv.height=Math.round(im.height*k);
   cv.getContext('2d').drawImage(im,0,0,cv.width,cv.height);
   const cvT=trimWhite(cv);
   enhanceCanvas(cvT);
   cb(cvT.toDataURL('image/jpeg',.9));
  };
  im.src=e.target.result;
 };
 r.readAsDataURL(file);
}
